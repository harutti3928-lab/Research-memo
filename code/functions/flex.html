FLEX — 自己エネルギー・感受率・自己無撞着ループ

— Green 関数・χ0・有効相互作用・Σ を FFT 畳み込みで回す FLEX コア  |  関数名をクリックで詳細 —

FLEX（Fluctuation Exchange Approximation）の自己無撞着ループの心臓部。 Nonint の H(k) と FFT の主要配列・プランを土台に、 G → χ0 → χs/χc → Veff → Σ を 1 反復として、μ をフィリング一定に 更新しながら Σ が収束するまで回す。畳み込み（χ0=GG、Σ=VeffG）は (r,τ) の要素積に落として FFT で往復する（kw2rt!/rt2kw!）。

calc_GREEN!→ calc_CHI0!→ calc_Veff!→ calc_SIGMA!→ calc_mu↻ 収束判定
このページの規約：フェルミオン Σ・G は正周波数 Nw 本＋折り返しで 2Nw。 G(-iω)=G†(iω) の関係で負周波数を埋める。ボゾン χ・Veff は 2Nw 本で iν=0 が先頭、Nw+1 が折り返し点（エルミート化）、負側は Veff(-iν)=Veff†(iν)。 軌道は複合添字 mask_inds（(mask-1)÷No+1, (mask-1)%No+1 で軌道ペアに展開）。 FFT は副作用を持つ：calc_CHI0! は G を (r,τ) 表示のまま残し、次段の calc_SIGMA! が それを前提に使う。数値ロジックは検証済み実装をそのまま移植。

# flex.jl
# FLEX（揺らぎ交換近似）の自己無撞着計算コア。
#   G(k,iω)  = [iω + μ - H(k) - Σ]^-1
#   χ0(q,iν) = -T/N Σ_k G G           （FFT で r,τ の積に落とす）
#   χ_s/χ_c  = χ0 (1 ∓ U χ0)^-1
#   Veff     = 3/2 U_s χ_s U_s + 1/2 U_c χ_c U_c - 1/4(U_s+U_c)χ0(U_s+U_c) + ...
#   Σ(k,iω)  = T/N Σ Veff G           （FFT）→ 混合して更新
# μ はフィリングを保つよう毎反復更新する。

概要

matsubara_fermion(Nw::Int, temperature::Real) = Array(im * π * temperature * range(1, 2 * Nw, step=2))

読み方

  • range(1, 2*Nw, step=2) — 1, 3, 5, …, 2Nw-1、すなわち 2n-1（n=1..Nw）の奇数列。フェルミオンは奇数×πT。
  • im * π * temperature * (…) — 各要素に iπT を掛けて iωn に。負周波数は G(-iω)=G†(iω) で後から埋めるので、ここでは正側 Nw 本だけ。
  • Array(...) — range の遅延オブジェクトを実体の配列に変換。

calc_GREEN!▶ G(k,iω) = [iω + μ - H(k) - Σ]-1 を各 k・各周波数で構築（in-place）
calc_GREEN!(G, SIGMA, ENERGY, MATSUBARA, mu) # 破壊的：G を書き込む

概要

Dyson 方程式の G(k,iω) = [iω + μ - H(k) - Σ(k,iω)]-1 を各 k・各正周波数 w で No×No 行列の逆行列として計算し、負周波数側は G(-iω)=G†(iω) で埋める。 SIGMA は正周波数 Nw 本、G は折り返し込みの 2Nw 本。

コード

flex.jl · calc_GREEN!
function calc_GREEN!(G::Array{ComplexF64,6}, SIGMA::Array{ComplexF64,6}, ENERGY::Array{ComplexF64,5}, MATSUBARA::Array{ComplexF64,1}, mu::Float64)
    Nw, Nqx, Nqy, Nqz, No, _ = size(SIGMA)
    Nq = Nqx * Nqy * Nqz
    SIGMA_view = reshape(SIGMA, Nw, Nq, No, No)
    ENERGY_view = reshape(ENERGY, Nq, No, No)
    G_view = reshape(G, 2 * Nw, Nq, No, No)

    dind = 1:No+1:No*No
    @threads for q in 1:Nq
        local_matrix = Array{ComplexF64}(undef, No, No)
        @views ENERGY_q = copy(ENERGY_view[q, :, :])
        for w in 1:Nw
            @views local_matrix .= -SIGMA_view[w, q, :, :] .- ENERGY_q
            local_matrix[dind] .+= MATSUBARA[w] + mu
            G_view[w, q, :, :] .= inv(local_matrix)
        end
        for j in 1:No, i in 1:No
            @views G_view[2*Nw:-1:Nw+1, q, i, j] .= conj(G_view[1:Nw, q, j, i])  # G(-iw) = G†(iw)
        end
    end
end

読み方

  • dind = 1:No+1:No*No — No×No 行列の対角の線形インデックス（1, No+2, 2No+3, …）。iω+μ を対角だけに足すために使う。
  • local_matrix .= -SIGMA .- ENERGY_q → local_matrix[dind] .+= MATSUBARA[w] + mu — -Σ-H を作り、対角に iωw+μ を加えて iω+μ-H-Σ にする。
  • G_view[w,q,:,:] .= inv(local_matrix) — 逆行列で G。各 k を @threads 並列、周波数 w は内側ループ。
  • G_view[2Nw:-1:Nw+1, q, i, j] .= conj(G_view[1:Nw, q, j, i]) — Gij(-iω) = conj(Gji(iω))。正周波数から負周波数を折り返しで生成（軌道 i,j も入れ替える点に注意）。
  • ENERGY_q = copy(ENERGY_view[q,:,:]) — その k の H(k) を一度コピーして周波数ループで使い回す（スライス参照の繰り返しを避ける）。
符号の由来： ENERGY（Nonint の H(k)）はエルミートなので -H と iω+μ を組んで逆行列を取れば G になる。Σ を 0 にすれば非相互作用 G0 と一致する（SCF 初回）。
関連：matsubara_fermion（MATSUBARA を供給）／ Nonint の ENERGY・SIGMA
calc_CHI0!▶ 既約感受率 χ0(q,iν) = -T/N Σk G G を FFT 畳み込みで（in-place・G を r,τ に残す）
calc_CHI0!(CHI, G, temperature, FFT_plan, mask_inds=nothing) # CHI←χ0(k,iν), G←G(r,τ)（副作用）

概要

既約（バブル）感受率 χ0(q,iν) = -T/N Σk G(k)G(k+q) を、(r,τ) での 要素積 G(r,τ)·G(-r,-τ) に落として計算する。kw2rt!(G) で G を (r,τ) に変換 → 積を CHI に書く → rt2kw!(CHI) で (q,iν) に戻す。戻ったあとも G は (r,τ) のまま （次段 calc_SIGMA! がそれを使う）。

コード

flex.jl · calc_CHI0!
function calc_CHI0!(CHI::Array{ComplexF64,6}, G::Array{ComplexF64,6}, temperature::Float64, FFT_plan, mask_inds=nothing)
    Nw2, Nqx, Nqy, Nqz, No, _ = size(G)
    Nw = Nw2 ÷ 2
    Nq = Nqx * Nqy * Nqz
    mask_inds = (mask_inds === nothing) ? (1:No*No) : mask_inds
    Nco = length(mask_inds)
    coef = -temperature / Nq

    CHI_view = reshape(CHI, 2 * Nw, Nq, Nco, Nco)
    G_view = reshape(G, 2 * Nw, Nq, No, No)

    kw2rt!(G, FFT_plan)
    i1_list = j1_list = (mask_inds .- 1) .÷ No .+ 1
    i2_list = j2_list = (mask_inds .- 1) .% No .+ 1
    x_, y_, z_, w_ = map(N -> [1; N:-1:2], (Nqx, Nqy, Nqz, 2 * Nw))
    xyz_ind2sub = CartesianIndices(Array{Int}(undef, Nqx, Nqy, Nqz))
    for jj in 1:Nco, ii in 1:Nco
        i1, i2 = i1_list[ii], i2_list[ii]
        j1, j2 = j1_list[jj], j2_list[jj]
        @threads for q in 1:Nq
            x, y, z = Tuple(xyz_ind2sub[q])
            xx, yy, zz = x_[x], y_[y], z_[z]
            for w in 1:Nw2
                CHI_view[w, q, ii, jj] = coef * G_view[w, q, i1, j1] * G[w_[w], xx, yy, zz, j2, i2]
            end
        end
    end
    rt2kw!(CHI, FFT_plan)
end

読み方

  • coef = -temperature / Nq — -T/N の前係数。
  • kw2rt!(G, FFT_plan) — G を (k,iω)→(r,τ) にその場変換。以降 G は (r,τ)。
  • i1_list = (mask_inds .- 1) .÷ No .+ 1 / i2_list = (mask_inds .- 1) .% No .+ 1 — 複合添字を軌道ペア (i1,i2) に展開（÷=商, %=余り）。mask_inds で計算する軌道成分を絞れる。
  • x_,y_,z_,w_ = map(N -> [1; N:-1:2], …) — r→-r と τ→-τ の反転インデックス。2Nw 軸も反転対象。
  • CHI = coef * G[w,q,i1,j1] * G[w_[w], -r, j2,i2] — χ0(r,τ) ∝ G(r,τ)·G(-r,-τ) の要素積。第 2 因子が (-r,-τ)（xx,yy,zz,w_[w]）になっているのがバブルの畳み込み。
  • rt2kw!(CHI, FFT_plan) — できた χ0(r,τ) を (q,iν) に戻す。
副作用に注意： この関数を抜けたとき G は (r,τ) 表示のまま残る（元の (k,iω) には戻らない）。これは意図的で、次の calc_SIGMA! が G(r,τ) をそのまま使って Σ=VeffG の積を作るため。順番を入れ替えると壊れる。
関連：kw2rt!/rt2kw!（FFT 往復）／ calc_Veff!（この χ0 を受ける）
calc_Veff!▶ χ0 から RPA 感受率 χs/χc を作り、有効相互作用 Veff を CHI に上書き
calc_Veff!(CHI, U_spin, U_charge; is_Uchi0_scale=false, maxeigval_tol=0.99) # CHI: χ0 → Veff

概要

既約感受率 χ0 から RPA のスピン/電荷感受率 χs=χ0(1-Usχ0)-1、 χc=χ0(1+Ucχ0)-1 を組み、 有効相互作用 Veff を計算して同じ CHI に上書きする（メモリ節約）。 ボゾン周波数の非負側 1:Nw+1 だけ計算し、負側は Veff(-iν)=Veff†(iν) で埋める。

コード

flex.jl · calc_Veff!
function calc_Veff!(CHI::Array{ComplexF64,6}, U_spin::Array{Float64,2}, U_charge::Array{Float64,2}; is_Uchi0_scale::Bool=false, maxeigval_tol::Float64=0.99)
    Nw2, Nqx, Nqy, Nqz, Nco, _ = size(CHI)
    Nw = Nw2 ÷ 2
    Nq = Nqx * Nqy * Nqz
    CHI_view = reshape(CHI, Nw2, Nq, Nco, Nco)

    U_spin_scale, U_charge_scale = 1.5 .* U_spin, 0.5 .* U_charge
    U_cs = (U_spin .+ U_charge) .* 0.5
    U_term = -0.5 .* U_charge .+ 1.5 .* U_spin
    dind = diagind(U_cs)

    @threads for q in 1:Nq
        chi_S = Array{ComplexF64}(undef, Nco, Nco)
        chi_C = Array{ComplexF64}(undef, Nco, Nco)
        tmp = Array{ComplexF64}(undef, Nco, Nco)
        chi0 = Array{ComplexF64}(undef, Nco, Nco)
        for w in 1:Nw+1
            @views chi0 .= CHI_view[w, q, :, :]

            # chi_S = chi0 (1 - U_S chi0)^-1
            tmp .= -U_spin * chi0
            if is_Uchi0_scale
                maxeigval = maximum(real.(eigvals(-tmp)))
                maxeigval > maxeigval_tol && (tmp .*= maxeigval_tol / maxeigval)
            end
            tmp[dind] .+= 1.0
            chi_S .= chi0
            rdiv!(chi_S, lu(tmp))

            # chi_C = chi0 (1 + U_C chi0)^-1
            tmp .= U_charge * chi0
            if is_Uchi0_scale
                maxeigval = maximum(real.(eigvals(-tmp)))
                maxeigval > maxeigval_tol && (tmp .*= maxeigval_tol / maxeigval)
            end
            tmp[dind] .+= 1.0
            chi_C .= chi0
            rdiv!(chi_C, lu(tmp))

            # Veff = 3/2 U_S χ_S U_S + 1/2 U_C χ_C U_C - 1/4(U_S+U_C)χ0(U_S+U_C) - 1/2 U_C + 3/2 U_S
            CHI_view[w, q, :, :] .= (U_spin_scale * chi_S * U_spin) .+
                                    (U_charge_scale * chi_C * U_charge) .+
                                    U_term .- (U_cs * chi0 * U_cs)
        end
        # Nw+1 には (Veff(iν) + Veff(-iν))/2 を入れる
        @views tmp .= CHI_view[Nw+1, q, :, :]
        CHI_view[Nw+1, q, :, :] .+= tmp'
        CHI_view[Nw+1, q, :, :] .*= 0.5
        for jj in 1:Nco, ii in 1:Nco
            # 残りは Veff(-iν) = Veff†(iν)
            @views CHI_view[2*Nw:-1:Nw+2, q, ii, jj] .= conj(CHI_view[2:Nw, q, jj, ii])
        end
    end
end

読み方

  • tmp .= -U_spin * chi0; tmp[dind] .+= 1.0 — 分母 1 - Usχ0 を組む。dind=diagind は複合行列の対角。
  • chi_S .= chi0; rdiv!(chi_S, lu(tmp)) — 右割り算 chi_S = χ0·tmp-1 = χ0(1-Usχ0)-1。lu で 1 回分解して解く（inv より安定・速い）。電荷側は符号が +。
  • is_Uchi0_scale ブロック — Usχ0 の最大固有値が maxeigval_tol(=0.99) を超えたら縮小。Stoner 不安定（Uχ0→1）で発散するのを防ぐ安全弁。磁気不安定の直前で数値を安定化する。
  • U_spin_scale=1.5Us, U_charge_scale=0.5Uc, U_term=1.5Us-0.5Uc — Veff の各係数を事前計算。
  • CHI_view[w,q,:,:] .= 3/2 UsχsUs + 1/2 UcχcUc + U_term - U_cs χ0 U_cs — Veff をその場で上書き。U_cs=(Us+Uc)/2。
  • Nw+1 の対称化 — (V+V†)/2 で折り返し点をエルミートにならす。
  • CHI_view[2Nw:-1:Nw+2, q, ii, jj] .= conj(CHI_view[2:Nw, q, jj, ii]) — 負ボゾン側を Veff(-iν)=Veff†(iν) で生成。
周波数の並びに注意： CHI（Veff）のボゾン軸は iν = 0, 2πTi, 4πTi, …（先頭 iν=0）で、1:Nw+1 が非負側、Nw+1 が折り返し点、Nw+2:2Nw が負側。ループ上限が Nw+1 なのはこのため。フェルミオン（Σ・G）の並びとは別物。
関連：calc_CHIS_CHIC!（χs/χc を別配列に保存する版）／ calc_SIGMA!（この Veff を使う）
calc_CHIS_CHIC!▶ χs/χc を別配列に保持（プロット・後処理用。Veff は作らない）
calc_CHIS_CHIC!(CHI_S, CHI_C, CHI0, U_spin, U_charge; is_Uchi0_scale=false, maxeigval_tol=0.99)

概要

calc_Veff! と同じ RPA 感受率の計算だが、Veff にまとめず χs=χ0(1-Usχ0)-1 と χc=χ0(1+Ucχ0)-1 をそれぞれ別配列 CHI_S・CHI_C に保存する。SCF には不要で、感受率のプロットや後処理のための版。

コード

function calc_CHIS_CHIC!(
    CHI_S::Array{ComplexF64,6}, CHI_C::Array{ComplexF64,6}, CHI0::Array{ComplexF64,6},
    U_spin::Array{Float64,2}, U_charge::Array{Float64,2};
    is_Uchi0_scale::Bool=false, maxeigval_tol::Float64=0.99,
)
    Nw2, Nqx, Nqy, Nqz, Nco, _ = size(CHI0)
    Nw = Nw2 ÷ 2
    Nq = Nqx * Nqy * Nqz

    CHI0_view = reshape(CHI0, Nw2, Nq, Nco, Nco)
    CHIS_view = reshape(CHI_S, Nw2, Nq, Nco, Nco)
    CHIC_view = reshape(CHI_C, Nw2, Nq, Nco, Nco)

    dind = diagind(U_spin)
    @threads for q in 1:Nq
        chi_S = Array{ComplexF64}(undef, Nco, Nco)
        chi_C = Array{ComplexF64}(undef, Nco, Nco)
        tmp = Array{ComplexF64}(undef, Nco, Nco)
        chi0 = Array{ComplexF64}(undef, Nco, Nco)
        for w in 1:Nw+1
            @views chi0 .= CHI0_view[w, q, :, :]

            tmp .= -U_spin * chi0
            if is_Uchi0_scale
                maxeigval = maximum(real.(eigvals(-tmp)))
                maxeigval > maxeigval_tol && (tmp .*= maxeigval_tol / maxeigval)
            end
            tmp[dind] .+= 1.0
            chi_S .= chi0
            rdiv!(chi_S, lu(tmp))
            @views CHIS_view[w, q, :, :] .= chi_S

            tmp .= U_charge * chi0
            if is_Uchi0_scale
                maxeigval = maximum(real.(eigvals(-tmp)))
                maxeigval > maxeigval_tol && (tmp .*= maxeigval_tol / maxeigval)
            end
            tmp[dind] .+= 1.0
            chi_C .= chi0
            rdiv!(chi_C, lu(tmp))
            @views CHIC_view[w, q, :, :] .= chi_C
        end

        # 中央周波数（iν=0 の折返し点）を対称化
        @views tmp .= CHIS_view[Nw+1, q, :, :]
        CHIS_view[Nw+1, q, :, :] .+= tmp'
        CHIS_view[Nw+1, q, :, :] .*= 0.5
        @views tmp .= CHIC_view[Nw+1, q, :, :]
        CHIC_view[Nw+1, q, :, :] .+= tmp'
        CHIC_view[Nw+1, q, :, :] .*= 0.5

        for jj in 1:Nco, ii in 1:Nco
            @views CHIS_view[2*Nw:-1:Nw+2, q, ii, jj] .= conj(CHIS_view[2:Nw, q, jj, ii])
            @views CHIC_view[2*Nw:-1:Nw+2, q, ii, jj] .= conj(CHIC_view[2:Nw, q, jj, ii])
        end
    end
    return nothing
end

読み方

  • 中身の χs/χc の作り方（tmp 組立 → rdiv!(…, lu(tmp)) → is_Uchi0_scale の固有値縮小）は calc_Veff! と完全に同一。
  • 違いは CHIS_view/CHIC_view に別々に格納する点だけ（Veff に畳まない）。
  • 折り返し点 Nw+1 の対称化と負側 2Nw:-1:Nw+2 ← conj(2:Nw) も同じ規約。
  • return nothing — 戻り値なし（配列を直接書き換える in-place）。
使い分け： SCF を回すだけなら calc_Veff!（Veff に上書きしてメモリ節約）。 スピン/電荷感受率そのものを図示・解析したいときは calc_CHIS_CHIC! で別配列に取り出す。数値ロジックは共通なので結果は整合する。
関連：calc_Veff!（同じ RPA を Veff に畳む版）／ calc_CHI0!（入力 χ0）
calc_SIGMA!▶ Σ = T/N Σ VeffG を FFT 畳み込みで作り、混合更新＋誤差評価（in-place）
calc_SIGMA!(CHI, G, SIGMA, temperature, FFT_plan; mask_inds, sub_self0=true, mix_factor=0.3, RPA=false, return_dself=true) → (abs_err, rel_err)

概要

自己エネルギー Σ13(r,τ) = T/N Σ24 V1234(r,τ) G24(r,τ) を (r,τ) の積で作り、rt2kw! で (k,iω) に戻して 混合更新 Σnew = mix·Σcal + (1-mix)·Σold する。入力は CHI=Veff(k,iν)・G=G(r,τ)（前段 calc_CHI0! の副作用）・SIGMA=Σold。 return_dself=true で (絶対誤差, 相対誤差) を返し収束判定に使う。

function calc_SIGMA!(CHI, G, SIGMA, temperature, FFT_plan;
                    mask_inds=nothing, sub_self0=true, mix_factor=0.3, RPA=false, return_dself=true)
    Nw, Nqx, Nqy, Nqz, No, _ = size(SIGMA)
    Nq = Nqx * Nqy * Nqz

    function _subtract_minimum_iw!(G_view)
        # 最低松原の実部を引く（第一原理バンド等で使用）。1:Nw のみ操作。
        self0 = Array{ComplexF64}(undef, Nq, No, No)
        for j in 1:No, i in 1:No
            @views self0[:, i, j] .= 0.5 * (G_view[1, :, i, j] + conj(G_view[1, :, j, i]))
        end
        for j in 1:No, i in 1:No
            @threads for q in 1:Nq
                G_view[1:Nw, q, i, j] .-= self0[q, i, j]
            end
        end
    end

    function _update_SIGMA!(G_view, SIGMA_view, mix_factor, RPA)
        for j in 1:No, i in 1:No
            @threads for q in 1:Nq
                if RPA
                    @views SIGMA_view[:, q, i, j] .= G_view[1:Nw, q, i, j]; continue
                end
                # G の Nw+1:2Nw は未使用なので、誤差 Σ(cal)-Σ(old) を格納
                @views G_view[2*Nw:-1:Nw+1, q, i, j] .= G_view[1:Nw, q, i, j] - SIGMA_view[:, q, i, j]
                # Σ(new) = mix*Σ(cal) + (1-mix)*Σ(old)
                SIGMA_view[:, q, i, j] .*= (1 - mix_factor)
                @views SIGMA_view[:, q, i, j] .+= mix_factor .* G_view[1:Nw, q, i, j]
            end
        end
    end

    function _eval_err(G_view, avoid_zero_div_factor=1e-10)
        a_err_list = Vector{Float64}(undef, Nq); r_err_list = Vector{Float64}(undef, Nq)
        @threads for q in 1:Nq
            dself_tmp = abs.(G_view[2*Nw:-1:Nw+1, q, :, :])
            a_err_list[q] = maximum(dself_tmp)
            dself_tmp ./= abs.(G_view[1:Nw, q, :, :]) .+ avoid_zero_div_factor
            r_err_list[q] = maximum(dself_tmp)
        end
        return maximum(a_err_list), maximum(r_err_list)
    end

    mask_inds = (mask_inds === nothing) ? (1:No*No) : mask_inds
    Nco = length(mask_inds)
    CHI_view = reshape(CHI, 2 * Nw, Nq, Nco, Nco)
    G_view = reshape(G, 2 * Nw, Nq, No, No)
    SIGMA_view = reshape(SIGMA, Nw, Nq, No, No)
    coef = temperature / Nq

    kw2rt!(CHI, FFT_plan)
    i1_list = j1_list = (mask_inds .- 1) .÷ No .+ 1
    i2_list = j2_list = (mask_inds .- 1) .% No .+ 1
    @threads for q in 1:Nq
        SIGMA_rt_tmp = zeros(ComplexF64, 2 * Nw, No, No)
        for jj in 1:Nco, ii in 1:Nco
            @views SIGMA_rt_tmp[:, i1_list[ii], j1_list[jj]] .+= CHI_view[:, q, ii, jj] .* G_view[:, q, i2_list[ii], j2_list[jj]]
        end
        G_view[:, q, :, :] .= SIGMA_rt_tmp .* coef
    end
    rt2kw!(G, FFT_plan)

    sub_self0 && _subtract_minimum_iw!(G_view)
    _update_SIGMA!(G_view, SIGMA_view, mix_factor, RPA)
    return_dself && return _eval_err(G_view)
end

読み方

  • kw2rt!(CHI) — Veff を (k,iν)→(r,τ) に変換。G は前段で既に (r,τ)。
  • SIGMA_rt_tmp[:, i1, j1] .+= CHI[:,q,ii,jj] .* G[:,q,i2,j2] — (r,τ) での要素積で Σ(r,τ)=T/N Σ VeffG。複合添字を軌道ペアに展開して和を取る。
  • G_view[:,q,:,:] .= SIGMA_rt_tmp .* coef → rt2kw!(G) — 計算した Σ(r,τ) を G のメモリに書き込み、(k,iω) に戻す。以後 G_view[1:Nw] が Σcal。
  • メモリ再利用の肝： G の未使用な上半分 Nw+1:2Nw に 誤差 Σcal-Σold を格納（_update_SIGMA!）。これを _eval_err がそのまま読んで絶対/相対誤差を出す。
  • _subtract_minimum_iw! — 最低松原成分の実部（self0）を Σ から引く。第一原理バンドで μ と二重にシフトしないための静的補正（1:Nw のみ）。sub_self0=true のとき実行。
  • _update_SIGMA! — Σnew=(1-mix)Σold+mix·Σcal の線形混合で SCF を安定化（既定 mix=0.3）。RPA=true なら混合せず Σ=Σcal。
  • return_dself && return _eval_err(...) — (絶対誤差, 相対誤差) を返す。呼び出し側はこれで収束を判定。
入力配列の状態が前提： G は (r,τ)（calc_CHI0! の副作用）でなければならない。関数内で G は「Σ の作業領域＋誤差バッファ」に転用され、calc_GREEN! を呼ぶまで (k,iω) の G としては使えなくなる。SCF の呼び出し順（G→χ0→Veff→Σ→次のG）を守ること。
関連：calc_Veff!（CHI=Veff を供給）／ calc_CHI0!（G を (r,τ) に）／ calc_GREEN!（更新後の Σ で次の G）
calc_mu▶ フィリング一定になる化学ポテンシャル μ を求根（Secant→Brent フォールバック）
calc_mu(mu, SIGMA, ENERGY, MATSUBARA, ENERGY_diag, temperature, Nfilling; solver="Secant", dmu=nothing, atol=1e-6, rtol=1e-8, show_Ne=false) → mu_new

概要

電子数 Ne(μ) が目標フィリング Nfilling に一致する μ を求根で決める。相互作用込みの G=[iω+μ-H-Σ]-1 の松原和から Ne(μ) を組み、Secant（割線法）で解いて、失敗時は Brent 法に自動フォールバックする。Ne は非相互作用のフェルミ分布項と、その相互作用補正（松原和）に 分けて評価し、収束を速める。

コード

flex.jl · calc_mu（抜粋：目標方程式とソルバ切替）
function calc_mu(mu, SIGMA, ENERGY, MATSUBARA, ENERGY_diag, temperature, Nfilling;
                solver="Secant", dmu=nothing, atol=1e-6, rtol=1e-8, show_Ne=false)
    Nw, Nqx, Nqy, Nqz, No, _ = size(SIGMA)
    Nq = Nqx * Nqy * Nqz
    SIGMA_view = reshape(SIGMA, Nw, Nq, No, No)
    ENERGY_view = reshape(ENERGY, Nq, No, No)

    dmu = (dmu === nothing) ? (maximum(ENERGY_diag) - minimum(ENERGY_diag)) * 0.5 : dmu
    coef1 = 2 * temperature / Nq
    coef2 = 1.0 / Nq
    beta = 1.0 / temperature

    Emin = minimum(ENERGY_diag); Emax = maximum(ENERGY_diag)
    # 数値不安定（NaN/Inf, secant の暴走）に対する μ の安全上限
    mu_abs_max = max(abs(Emin), abs(Emax)) + 100.0 * max(1.0, (Emax - Emin)) + 100.0

    function _target_equation(mu)
        dNe_kx = zeros(Nq)
        dind = 1:No+1:No*No
        @threads for q in 1:Nq
            local_matrix1 = Array{ComplexF64}(undef, No, No)
            local_matrix2 = Array{ComplexF64}(undef, No, No)
            @views ENERGY_q = copy(ENERGY_view[q, :, :])
            for w in 1:Nw
                local_matrix1 .= -ENERGY_q
                local_matrix1[dind] .+= MATSUBARA[w] + mu
                local_matrix2 .= -inv(local_matrix1)
                @views local_matrix1 .-= SIGMA_view[w, q, :, :]
                local_matrix2 .+= inv(local_matrix1)
                dNe_kx[q] += real(tr(local_matrix2))
            end
        end
        any(x -> !isfinite(x), dNe_kx) && throw(DomainError(mu, "calc_mu: non-finite dNe_kx"))
        Ne = sum(dNe_kx) * coef1
        Ne += sum(logistic.(-beta * (ENERGY_diag .- mu))) * coef2
        isfinite(Ne) || throw(DomainError(mu, "calc_mu: non-finite Ne"))
        show_Ne && @show mu, Ne
        return Ne - Nfilling
    end

    function _solve_Brent(mu, dmu)
        try
            return fzero(_target_equation, (mu - dmu, mu + dmu), Roots.Brent(), atol=atol, rtol=rtol)
        catch
            return "Failed-to-converge"
        end
    end

    if solver == "Secant"
        try
            mu_new = find_zero(_target_equation, mu, Order1(), atol=atol, rtol=rtol)
            if !(isfinite(mu_new) && abs(mu_new) <= mu_abs_max)
                @warn "calc_mu: Secant returned implausible mu; fallback to Brent" mu mu_new Emin Emax mu_abs_max
                mu_new = _solve_Brent(mu, dmu)
            end
        catch
            @warn "Failed convergence with the secant method and switched to the Brent method."
            mu_new = _solve_Brent(mu, dmu)
        end
    elseif solver == "Brent"
        mu_new = _solve_Brent(mu, dmu)
        if mu_new == "Failed-to-converge"
            dmu = (maximum(ENERGY_diag) - minimum(ENERGY_diag)) * 0.5
            mu_new = _solve_Brent(mu, dmu)
        end
    else
        throw(ArgumentError("solver must be either \"Secant\" or \"Brent\"."))
    end
    return mu_new
end

読み方

  • _target_equation(mu) — Ne(μ)-Nfilling を返す求根対象。ゼロになる μ が答え。
  • local_matrix2 .= -inv(iω+μ-H) → .+= inv(iω+μ-H-Σ) — 各周波数で「相互作用 G」と「非相互作用 G」の差のトレースを取る。dNe_kx[q] はその松原和。
  • Ne = sum(dNe_kx)*coef1（coef1=2T/N）— 差分の松原和（正周波数だけ回して ×2 で負側を補う）＝相互作用による占有数補正。
  • Ne += sum(logistic(-β(E_diag-μ)))*coef2 — 非相互作用バンドのフェルミ分布（logistic=フェルミ関数）を閉形式で加える。松原和が遅い分をこの解析項が肩代わりし、収束を速める定石。
  • mu_abs_max — secant が発散して非現実的な μ を返したときの安全上限。超えたら Brent に切替。
  • solver=="Secant" — find_zero(…, Order1())（割線法）。例外 or 実装外の値なら @warn して _solve_Brent に自動フォールバック。
  • solver=="Brent" — 区間 (mu-dmu, mu+dmu) を括る Brent 法。失敗時は dmu をバンド幅の半分に広げて再試行。
  • throw(DomainError(...)) — NaN/Inf が出たら黙って進めず即エラー。数値破綻を早期に検出。
Secant と Brent の使い分け： 割線法は初期 μ 1 点から速く収束するが暴走し得る。Brent は区間を括るぶん安全だが dmu（区間幅）が要る。既定は「まず Secant、危なければ Brent」の二段構え。dmu を明示するとバンド幅からの推定を省けて速い。
関連：calc_GREEN!（更新した μ で G を作り直す）／ Nonint の ENERGY_diag（フェルミ項に使用）
1 反復の流れ： calc_GREEN!（Σ,μ から G）→ calc_CHI0!（G→χ0、G は (r,τ) に）→ calc_Veff!（χ0→Veff）→ calc_SIGMA!（Veff,G→Σ を混合更新・誤差評価）→ calc_mu（フィリング一定に μ 更新）。これを誤差が閾値を切るまで繰り返す。感受率を図示したいときだけ calc_CHIS_CHIC! を差し込む。FFT の副作用（G の (r,τ) 化）と配列の使い回しが全体設計の要。
